import requests, time, base64, json

def run_test(scenario, video_path=None, json_data=None, system_prompt="", user_prompt=""):
  content = [{"type": "text", "text": user_prompt}]

  if video_path: # attach video frame as image (extract key frame first)
    with open(video_path, "rb") as f:
      b64 = base64.b64encode(f.read()).decode()
    content.insert(0, {"type": "image_url", "image_url": {"url": f"data:image/jpeg;base64,{b64}"}})

  t_start = time.time()
  resp = requests.post(BASE_URL, headers={"Authorization": f"Bearer {API_KEY}"},
    json={"model": MODEL, "messages": [{"role": "system", "content": system_prompt}, {"role": "user", "content": content}]})
  latency = round(time.time() - t_start, 2)

  return {"latency_s": latency, "output": resp.json()["choices"][0]["message"]["content"]}
